# Runtime dependencies of the desktop app, for the packaged build.
#
# Deliberately NOT requirements.txt: that file is a full `pip freeze` of the
# development venv on Linux, which pins build-only and platform-specific
# packages that either do not exist or are not wanted on a Windows build
# runner. This is the smaller set the GUI actually imports.
#
# Versions are pinned to what the app was built and verified against.
# cocoex is published as `coco-experiment`, which ships cp39-cp312
# win_amd64 wheels - hence the Python 3.10 pin in the build workflow.

PySide6==6.11.2
coco-experiment==2.8.2
opfunu==1.0.4

numpy==2.2.6
pandas==2.3.3
scipy==1.15.3
scikit-learn==1.7.2
matplotlib==3.10.8
seaborn==0.13.2

# yellowbrick + kneed: 03_cluster/cluster_trajectories.py's elbow-method k
# selection, which the Process tab runs in the packaged app now (see
# packaging/gui.spec's own note on why these two aren't excluded).
yellowbrick==1.5
kneed==0.8.6

# utils.py and several 03_cluster/04_metrics/05_analysis stage modules
# import this directly for progress bars. Not a dependency of anything else
# pinned here (verified: none of PySide6/coco-experiment/opfunu/numpy/
# pandas/scipy/scikit-learn/matplotlib/seaborn/fastparquet/yellowbrick/kneed
# declare it), so nothing pulls it in on its own - omitting it produces a
# working *build* (PyInstaller has nothing to complain about) that fails at
# *runtime* the moment the Setup tab tries to discover optimizers, with
# "No module named tqdm".
tqdm==4.67.3

# Parquet reader for data/clustering_latest/clustering_results/*.parquet.
# fastparquet rather than pyarrow: it reads the same files and is ~137 MB
# smaller in the bundle (see packaging/gui.spec).
fastparquet==2026.3.0

pyinstaller==6.22.3

# mealpy==3.0.3 is NOT listed above on purpose - it declares
# numpy<=1.26.0,>=1.17.1 (still true of 3.0.3, the latest release on PyPI),
# which conflicts with numpy==2.2.6 above and makes `pip install -r` on this
# file fail outright with ResolutionImpossible. That upper bound is stale,
# not a real incompatibility: `pip check` in the dev venv reports the exact
# same conflict, and mealpy has been running real optimizations against
# numpy 2.2.6 there the whole time (see gui/self_test.py's "a real
# optimization run completes" check, which is the actual verification that
# matters here, not the package metadata). Install it separately:
#   pip install --no-deps mealpy==3.0.3
# so pip never has to resolve its stale numpy pin against everything else.
